// Chart linking: binds the primary chart to comparison panes as one
// workspace. Pure chrome — no backend involved.
import { createLinkGroup, LinkGroup } from "openalgo-charts";

export interface LinkHost {
  charts(): unknown[]; // [primary, ...comparisonCharts]
  subscribeCharts(cb: (charts: unknown[]) => void): () => void;
}

let group: LinkGroup | null = null;

export function ensureLinkGroup(host: LinkHost): LinkGroup {
  if (group) return group;
  group = createLinkGroup({ crosshair: true, viewport: true, symbol: false, whenMissing: "nearest" });
  host.subscribeCharts((charts) => {
    group?.unlink();
    for (const c of charts) group?.add(c as never);
  });
  return group;
}

export function unlinkAll(): void {
  group?.unlink();
}
```

// Keyboard shortcuts: ShortcutManager maps keys to commands; this module
// routes those commands onto the shell's existing actions. Pure chrome.
import { ShortcutManager, DEFAULT_KEYMAP } from "openalgo-charts";

export interface ShortcutActions {
  panLeft(): void; panRight(): void;
  zoomIn(): void; zoomOut(): void;
  resetScale(): void; fitContent(): void;
  screenshot(): void;
  toggleGridVert(): void; toggleGridHorz(): void;
  [command: string]: (() => void) | undefined;
}

export function createShellShortcuts(actions: ShortcutActions): { manager: ShortcutManager; start(): () => void } {
  const manager = new ShortcutManager({ scope: "hover", persist: false });
  const off = manager.on((e) => {
    const fn = actions[e.command];
    if (fn) fn();
  });
  return {
    manager,
    start: () => {
      const handler = (ev: KeyboardEvent) => {
        const cmd = manager.resolve(ev as never);
        if (cmd) { ev.preventDefault(); actions[cmd]?.(); }
      };
      window.addEventListener("keydown", handler);
      return () => window.removeEventListener("keydown", handler);
    },
  };
}

export { DEFAULT_KEYMAP };
```
NOTE: `scope: "hover"` vs `"global"` — the reference default is hover (only fires when the mouse is over the chart). For a terminal shell, `global` may be more useful — decide and document. `manager.resolve(ev)` needs a `KeyLike` — check the type and pass the KeyboardEvent (cast as needed).
