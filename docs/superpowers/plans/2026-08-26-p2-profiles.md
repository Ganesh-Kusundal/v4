```python
def compute_profile(pid: str, bars: list[dict[str, Any]], params: dict[str, Any]) -> dict[str, Any]:
    spec = PROFILE_SPECS.get(pid)
    if spec is None:
        raise ValueError(f"unknown profile id: {pid!r}")
    # Coerce params per spec default type (int/float/str), mirroring P1.
    kwargs = {}
    for key, default in spec["params"].items():
        value = params.get(key, default)
        if isinstance(default, bool):
            kwargs[key] = bool(value)
        elif isinstance(default, int):
            kwargs[key] = int(value)
        elif isinstance(default, float):
            kwargs[key] = float(value)
        else:
            kwargs[key] = value
    return spec["function"](bars, **kwargs)
```

```python
    async def compute_profile_endpoint(profile_id: str, body: dict) -> dict:
        from tradex_trading.analytics.profiles import compute_profile

            else:
```

```ts
// computed by tradex_trading (POST /api/charts/profiles/{id}); this module
// draws what the backend returns. No profile math in JS.
import type { IPrimitive, PrimitiveHost, Bar } from "openalgo-charts";

export interface ProfilePayload { id: string; result: unknown; }

export async function computeProfile(id: string, params: Record<string, unknown>, bars: Bar[]): Promise<unknown> {
  const resp = await fetch(`/api/charts/profiles/${id}`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ id, params, bars }),
  });
  if (!resp.ok) throw new Error(`profile ${id} failed: ${await resp.text()}`);
  const body = (await resp.json()) as ProfilePayload;
  return body.result;
}
```

```ts
// renders it. This is the 91st indicator — the reference's one remaining

import type { IPrimitive, PrimitiveHost, Bar } from "openalgo-charts";

export interface SeasonalityTable {
  rows: { text: string; bgColor?: string }[][];
  options: { position: string; cellWidth: number[]; cellHeight: number; rowWeights: number[]; widthPercent: number; heightPercent: number; fontSize: number; margin: number };
}

export async function computeSeasonality(bars: Bar[], startYear = 2015): Promise<SeasonalityTable> {
  const resp = await fetch("/api/charts/profiles/seasonality", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ id: "seasonality", params: { start_year: startYear }, bars }),
  });
  if (!resp.ok) throw new Error(`seasonality failed: ${await resp.status}`);
  const body = (await resp.json()) as { result: SeasonalityTable };

}
```
